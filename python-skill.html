# A simple rule-based chatbot — one of my first Python projects! 🐍 class HyphenBot: def __init__(self): self.name = "HyphenBot" self.responses = { "hello": "Hey! Nice to meet you 👋", "who made you": "Aaradhya did! 💜", "bye": "Goodbye! Come back soon!", } def respond(self, user_input): lower = user_input.lower() for key in self.responses: if key in lower: return self.responses[key] return "Hmm, I don't know that yet! 🤔" # Run the bot bot = HyphenBot() print(bot.respond("hello")) # Hey! Nice to meet you 👋 print(bot.respond("who made you")) # Aaradhya did! 💜
